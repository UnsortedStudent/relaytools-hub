n8n Workflow Automation

n8n Apify Node & HTTP Template

Install the official Apify node in n8n (or paste this zero-dependency HTTP Request node) to trigger any Actor on a cron schedule and push JSON records to Slack, HubSpot, or Google Sheets:

n8n-relaytools-workflow.json 
{
  "name": "RelayTools CVE & Email Security Daily Alert (n8n)",
  "nodes": [
    {
      "name": "Daily Schedule",
      "type": "n8n-nodes-base.scheduleTrigger",
      "parameters": {
        "rule": {
          "interval": [
            {
              "field": "days"
            }
          ]
        }
      }
    },
    {
      "name": "Run Apify Actor (Sync Dataset)",
      "type": "n8n-nodes-base.httpRequest",
      "parameters": {
        "method": "POST",
        "url": "https://api.apify.com/v2/acts/regenerative_cornice~cve-kev-security-intel/run-sync-get-dataset-items",
        "authentication": "genericCredentialType",
        "sendHeaders": true,
        "headerParameters": {
          "parameters": [
            {
              "name": "Authorization",
              "value": "Bearer {{$env.APIFY_TOKEN}}"
            }
          ]
        },
        "sendBody": true,
        "specifyBody": "json",
        "jsonBody": {
          "mode": "kev_recent",
          "maxItems": 15,
          "ransomwareOnly": false
        }
      }
    }
  ]
}
